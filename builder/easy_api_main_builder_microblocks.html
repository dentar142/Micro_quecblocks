# -*- coding: utf-8 -*-
import easy_api as api

# 参数设置
DISPLAY_CYCLES = None
BUTTON_POLL_MS = 20
SENSOR_REFRESH_MS = 1000
UART_ID = 2                 # fixed UART2 / USART_B, PC COM43
UART_BAUDRATE = 115200
UART_TIMEOUT_MS = 1000

# 启用或关闭模块
api.init()                 # initialize easy_api
api.anjian(1)
api.guangmin(1)
api.i2c(1)
api.wenhumi(1)
api.jiasudu(1)
api.gpio(1)
api.pwm(1)
api.lcd(1)
api.uart(1)
api.cunchu(1)
api.yinpin(1)
api.ble(1, "server", "Uniknect_BLE_DEMO")
api.startpwm("D3", 1000, 50)
api.setuart(UART_ID, UART_BAUDRATE, UART_TIMEOUT_MS)


def main():
    loop = 0
    sensor_timer = 0
    key_adc_at_press = {}
    pwm_duty = 50
    recording = False
    play_message = "Playing..."

    while DISPLAY_CYCLES is None or loop < DISPLAY_CYCLES:
        key_event = api.readanjian()
        key_status = api.keytext(key_event)
        recording = api.updateaudio()
        uart_received_text = api.readuarttext()
        ble_received_text = api.readbledata()
        row6_message_active = api.lcdrowtempactive(5)
        api.updatelcdtemp()
        if key_status is not None:
            api.clearlcdline(4)
            api.showlcdcolor(key_status, 4, 0, "yellow")
        if uart_received_text is not None:
            api.showlcdrowtemp(uart_received_text, 5000, 3, 0, "cyan")
        if ble_received_text is not None:
            api.showlcdrowtemp(ble_received_text, 5000, 3, 0, "magenta")
        if key_event and key_event[0] == "user" and key_event[1] == "press":
            api.clearlcdline(5)
            api.showlcdcolor("Recording...", 5, 0, "red")
            api.recordtimed("SD:competition_test.wav", 5000)
            recording = True
        if key_event and key_event[0] == "center" and key_event[1] == "press":
            api.showlcdrowtemp(play_message, 5000, 5, 0, "green")
            api.playfile("SD:competition_test.wav", False)
        if key_event and key_event[0] == "left" and key_event[1] == "press":
            if pwm_duty > 0:
                pwm_duty += -5
                api.setpwmduty(pwm_duty)
        if key_event and key_event[0] == "left" and key_event[1] == "repeat":
            if pwm_duty > 0:
                pwm_duty += -5
                api.setpwmduty(pwm_duty)
        if key_event and key_event[0] == "right" and key_event[1] == "press":
            if pwm_duty < 100:
                pwm_duty += 5
                api.setpwmduty(pwm_duty)
        if key_event and key_event[0] == "right" and key_event[1] == "repeat":
            if pwm_duty < 100:
                pwm_duty += 5
                api.setpwmduty(pwm_duty)
        sensor_timer += BUTTON_POLL_MS

        if sensor_timer >= SENSOR_REFRESH_MS:
            sensor_timer = 0
            loop += 1
            light_value = api.readguangmin()
            light_value_text = api.baoliuxiaoshu(light_value, 0)
            light_percent = api.readguangmin_percent()
            light_percent_text = api.baoliuxiaoshu(light_percent, 0)
            wendu = api.readwendu()
            wendu_text = api.baoliuxiaoshu(wendu, 1)
            shidu = api.readshidu()
            shidu_text = api.baoliuxiaoshu(shidu, 1)
            x = api.readx()
            x_text = api.baoliuxiaoshu(x, 1)
            y = api.readyaxis()
            y_text = api.baoliuxiaoshu(y, 1)
            z = api.readz()
            z_text = api.baoliuxiaoshu(z, 1)
            light_lcd_text = api.pin_jie("L:", light_value_text, " P:", light_percent_text, "%", "")
            api.clearlcdline(0)
            api.showlcdcolor("T:", 0, 0, "red")
            api.showlcdcolor(wendu_text, 0, 2, "red")
            api.showlcdcolor("H:", 0, 9, "blue")
            api.showlcdcolor(shidu_text, 0, 11, "blue")
            api.clearlcdline(1)
            api.showlcdcolor(light_lcd_text, 1, 0, "white")
            api.clearlcdline(2)
            api.showlcdcolor("X:", 2, 0, "red")
            api.showlcdcolor(x_text, 2, 2, "red")
            api.showlcdcolor("Y:", 2, 7, "green")
            api.showlcdcolor(y_text, 2, 9, "green")
            api.showlcdcolor("Z:", 2, 14, "blue")
            api.showlcdcolor(z_text, 2, 16, "blue")
            if recording == False:
                if row6_message_active == False:
                    high_pin_text = api.highpins(("PA3", "PC0", "PC3", "PA5", "PA6", "PF13", "PF14", "PF15"))
                    api.clearlcdline(5)
                    api.showlcdcolor(high_pin_text, 5, 0, "white")
            climate_text = api.pin_jie("T:", wendu_text, " H:", shidu_text, "", "")
            light_text = api.pin_jie("L:", light_value_text, " P:", light_percent_text, "%", "")
            axis_text = api.pin_jie("X:", x_text, " Y:", y_text, " Z:", z_text)
            ble_x_text = api.pin_jie("X:", x_text, "", "", "", "")
            ble_y_text = api.pin_jie("Y:", y_text, "", "", "", "")
            ble_z_text = api.pin_jie("Z:", z_text, "", "", "", "")
            sensor_text = api.pin_jie(climate_text, " ", light_text, " ", axis_text, "\r\n")
            api.senduart(sensor_text)
            api.sendble(climate_text)
            api.sendble(light_text)
            api.sendble(ble_x_text)
            api.sendble(ble_y_text)
            api.sendble(ble_z_text)

        api.delay(BUTTON_POLL_MS)


if __name__ == "__main__":
    main()
